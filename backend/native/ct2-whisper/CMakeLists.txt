cmake_minimum_required(VERSION 3.16)
project(nastool_ct2_whisper LANGUAGES C CXX)
find_path(AVUTIL_INCLUDE_DIR libavutil/tx.h REQUIRED)
find_library(AVUTIL_LIBRARY avutil REQUIRED)
if(CT2_SOURCE)
  add_subdirectory("${CT2_SOURCE}" ctranslate2 EXCLUDE_FROM_ALL)
  set(CT2_LIBRARY ctranslate2)
else()
  find_package(ctranslate2 CONFIG REQUIRED)
  set(CT2_LIBRARY CTranslate2::ctranslate2)
endif()
add_executable(nastool-ct2-whisper main.cc)
target_compile_features(nastool-ct2-whisper PRIVATE cxx_std_17)
target_include_directories(nastool-ct2-whisper PRIVATE ${AVUTIL_INCLUDE_DIR})
target_link_libraries(nastool-ct2-whisper PRIVATE ${CT2_LIBRARY} ${AVUTIL_LIBRARY})
